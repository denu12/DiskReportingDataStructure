cmake_minimum_required(VERSION 3.20)
project(diskreport_ddim LANGUAGES CXX)
add_library(diskreport_ddim INTERFACE)
target_include_directories(diskreport_ddim INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(diskreport_ddim INTERFACE cxx_std_20)
add_executable(ddim_check check_campaign.cpp)
target_link_libraries(ddim_check PRIVATE diskreport_ddim)
target_include_directories(ddim_check SYSTEM PRIVATE ../../third_party/stann/include)

add_executable(morton_ddim_check morton_campaign.cpp)
target_link_libraries(morton_ddim_check PRIVATE diskreport_ddim)
target_compile_options(morton_ddim_check PRIVATE -O3 -mavx2)

add_executable(morton_3d_check morton_3d_campaign.cpp)
target_link_libraries(morton_3d_check PRIVATE diskreport_ddim)
target_compile_options(morton_3d_check PRIVATE -O3 -mavx2)

include(CTest)
if(BUILD_TESTING)
    add_executable(morton_ddim_test ../../morton/morton_d_dim_test.cc)
    target_compile_features(morton_ddim_test PRIVATE cxx_std_20)
    target_compile_options(morton_ddim_test PRIVATE -mavx2)
    add_test(NAME morton_ddim_test COMMAND morton_ddim_test)
endif()

add_executable(morton_plus_3d_check morton_plus_3d_campaign.cpp)
target_link_libraries(morton_plus_3d_check PRIVATE diskreport_ddim)
target_compile_options(morton_plus_3d_check PRIVATE -O3 -mavx2)
if(BUILD_TESTING)
    add_executable(mortonplus3d_test ../../morton/mortonplus3d_test.cc)
    target_compile_features(mortonplus3d_test PRIVATE cxx_std_20)
    target_compile_options(mortonplus3d_test PRIVATE -mavx2)
    find_package(Threads REQUIRED)
    target_link_libraries(mortonplus3d_test PRIVATE Threads::Threads)
    add_test(NAME mortonplus3d_test COMMAND mortonplus3d_test)
endif()

add_executable(morton_plus_ddim_check morton_plus_ddim_campaign.cpp)
target_link_libraries(morton_plus_ddim_check PRIVATE diskreport_ddim)
target_compile_options(morton_plus_ddim_check PRIVATE -O3 -mavx2)
if(BUILD_TESTING)
    add_executable(mortonplus_ddim_test ../../morton/mortonplus_d_dim_test.cc)
    target_compile_features(mortonplus_ddim_test PRIVATE cxx_std_20)
    target_compile_options(mortonplus_ddim_test PRIVATE -mavx2)
    add_test(NAME mortonplus_ddim_test COMMAND mortonplus_ddim_test)
endif()
